! =====================================
! R2 Router Configuration
! LAN: 192.168.1.128/27 (18 hosts)
! WAN1: 192.168.1.172/30 (R1-R2 link)
! WAN2: 192.168.1.176/30 (R2-R3 link)
! =====================================

enable
configure terminal
hostname R2

! Configure LAN interface (18 hosts)
interface FastEthernet0/0
ip address 192.168.1.129 255.255.255.224
description "LAN2 - 18 hosts"
no shutdown
exit

! Configure WAN interface to R1
interface Serial0/0/0
ip address 192.168.1.174 255.255.255.252
description "WAN Link to R1"
no shutdown
exit

! Configure WAN interface to R3
interface Serial0/0/1
ip address 192.168.1.177 255.255.255.252
description "WAN Link to R3"
clock rate 64000
no shutdown
exit

! Static routes to all remote networks
! Route to R1 LAN (64 hosts)
ip route 192.168.1.0 255.255.255.128 192.168.1.173

! Route to R3 LAN (4 hosts)
ip route 192.168.1.160 255.255.255.248 192.168.1.178

! Route to R4 LAN (2 hosts)
ip route 192.168.1.168 255.255.255.252 192.168.1.178

! Route to R3-R4 WAN link
ip route 192.168.1.180 255.255.255.252 192.168.1.178

! Enable CDP for troubleshooting
cdp run

! Save configuration
exit
copy running-config startup-config

! =====================================
! Configuration Complete
! Test with: ping 192.168.1.2 (PC1)
! =====================================
