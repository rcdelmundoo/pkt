! =====================================
! R1 Router Configuration
! LAN: 192.168.1.0/25 (64 hosts)
! WAN: 192.168.1.172/30 (R1-R2 link)
! =====================================

enable
configure terminal
hostname R1

! Configure LAN interface (64 hosts)
interface FastEthernet0/0
ip address 192.168.1.1 255.255.255.128
description "LAN1 - 64 hosts"
no shutdown
exit

! Configure WAN interface to R2
interface Serial0/0/0
ip address 192.168.1.173 255.255.255.252
description "WAN Link to R2"
clock rate 64000
no shutdown
exit

! Static routes to all remote networks
! Route to R2 LAN (18 hosts)
ip route 192.168.1.128 255.255.255.224 192.168.1.174

! Route to R3 LAN (4 hosts)
ip route 192.168.1.160 255.255.255.248 192.168.1.174

! Route to R4 LAN (2 hosts)
ip route 192.168.1.168 255.255.255.252 192.168.1.174

! Route to R2-R3 WAN link
ip route 192.168.1.176 255.255.255.252 192.168.1.174

! Route to R3-R4 WAN link
ip route 192.168.1.180 255.255.255.252 192.168.1.174

! Enable CDP for troubleshooting
cdp run

! Save configuration
exit
copy running-config startup-config

! =====================================
! Configuration Complete
! Test with: ping 192.168.1.130 (PC2)
! =====================================
